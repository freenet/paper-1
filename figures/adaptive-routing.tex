% Adaptive routing: per-neighbor latency predictions of the deployed
% hierarchical estimator (freenet-core router/hierarchical.rs). Latency is
% modeled on a log scale, so each neighbor's prediction is the shared isotonic
% distance curve multiplied by a neighbor-specific factor (its shrunk offset in
% log space); the per-arc offset is not drawn. The shared curve's fitted
% points (dots) are common to every neighbor. Beyond the fitted range the
% log-scale curves hold their end value rather than extrapolating, so the
% dashed segments are flat. Drawn schematically.
\begin{figure}[t]
\centering
\begin{tikzpicture}[
    axis/.style={->, >=Stealth, thick, black!75},
    curveA/.style={blue!65!black, thick},
    curveB/.style={orange!75!black, thick},
    curveC/.style={green!55!black, thick},
    fitdot/.style={circle, fill=white, draw=black, thin, inner sep=1.1pt},
    pickdot/.style={circle, fill, inner sep=1.2pt},
    label/.style={font=\scriptsize},
    >=Stealth,
]
  \draw[axis] (0,0) -- (7.0,0) node[right, label] {distance to target $d$};
  \draw[axis] (0,0) -- (0,3.6) node[above, label, align=center]
       {predicted\\latency $\widehat{\tau}_q(d)$};

  % Neighbor q_A: shared curve scaled by 1.0.
  \draw[curveA] (0.5,0.60) -- (1.4,0.75) -- (2.6,0.95) -- (3.8,1.15) -- (5.0,1.40);
  \draw[curveA, dashed] (0.0,0.60) -- (0.5,0.60);
  \draw[curveA, dashed] (5.0,1.40) -- (6.7,1.40);
  \node[label, text=blue!65!black, anchor=west] at (6.75,1.40) {$\widehat{\tau}_{q_A}$ (fast)};

  % Neighbor q_B: shared curve scaled by 1.6.
  \draw[curveB] (0.5,0.96) -- (1.4,1.20) -- (2.6,1.52) -- (3.8,1.84) -- (5.0,2.24);
  \draw[curveB, dashed] (0.0,0.96) -- (0.5,0.96);
  \draw[curveB, dashed] (5.0,2.24) -- (6.7,2.24);
  \node[label, text=orange!75!black, anchor=west] at (6.75,2.24) {$\widehat{\tau}_{q_B}$ (average)};

  % Neighbor q_C: shared curve scaled by 2.2.
  \draw[curveC] (0.5,1.32) -- (1.4,1.65) -- (2.6,2.09) -- (3.8,2.53) -- (5.0,3.08);
  \draw[curveC, dashed] (0.0,1.32) -- (0.5,1.32);
  \draw[curveC, dashed] (5.0,3.08) -- (6.7,3.08);
  \node[label, text=green!55!black, anchor=west] at (6.75,3.08) {$\widehat{\tau}_{q_C}$ (slow)};

  \foreach \x/\y in {0.5/0.6, 1.4/0.75, 2.6/0.95, 3.8/1.15, 5.0/1.4}

  \draw[gray!60, thin, dashed] (3.0,0) -- (3.0,3.4)

  \node[pickdot, orange!75!black] at (3.0,1.627) {};
  \node[pickdot, green!55!black] at (3.0,2.237) {};

\caption{Per-neighbor latency predictions used by adaptive routing.
Each neighbor $q$'s prediction $\widehat{\tau}_q(d)$ is a non-decreasing
function of ring distance to the target: one distance curve, fit by isotonic
regression (pool-adjacent-violators) over all neighbors' routing events, scaled
for each neighbor by a factor that is shrunk toward the network-wide value in
proportion to how little evidence supports it. Dots mark the shared fitted
points; beyond the observed distances each curve holds its end value. Two
further predictions track failure probability and throughput; one outcome is
shown for clarity. For a request to a target at distance $d_t$, the router
reads each neighbor's predictions at $d_t$ and picks the neighbor that
minimizes the combined expected cost. The chosen neighbor need not be the one
topologically closest to the target.}
\label{fig:adaptive-routing}
\end{figure}
